\begin{thm}[Descantes]
    Sei $f(x)=a_nx^n+a_{n-1}x^{n-1}+\hdots+a_1x+a_0\in\Z[x]$\\
    und sei $x_0=\frac{r}{s}$ eine Lösung von $f(x)=0$.
    Dann ist $r\in\Z$ ein Teiler von $a_0$ und $s\in\Z$
    ist ein Teiler von $a_n$\\
    Beweis: Sei $x_0=\frac{r}{s}$, $f(x_0)=0$
    \begin{align*}
        f(x_0)&=0\\
        \Rightarrow
        a_n\left(\frac{r}{s}\right)^n+\hdots+a_0&=0&|\cdot s^n\\
        \Rightarrow
        a_nr^n+a_{n-1}r^{n-1}s+\hdots+a_1rs^{n-1}+a_0s^n&=0\\
        \Rightarrow
        r(a_nr^{n-1}+\hdots+a_1s^{n-1})&=-a_0s^n
    \end{align*}
    $r,s$ sind Teilerfremd, aber
    \begin{equation*}
        (r|-a_0s^n)\Rightarrow r|a_0
    \end{equation*}
    $s$ ausklammern:
    \begin{equation*}
        (s|-a_nr^n)\Rightarrow s|a_n
    \end{equation*}
\end{thm}

\begin{defi}
    Ein \textbf{Ring} $R$ ist ein Quintupel $(R,+,\cdot,0,1)$
    bestehend aus eine Menge $R$ mit zwei Abbildungen
    \begin{itemize}
        \item
            Addition:
               $+:R\times R\rightarrow R,(a,b)\mapsto a+b$
        \item
            Multiplikation:
               $\cdot:R\times R\rightarrow R,(a,b)
            \mapsto a\cdot b$
    \end{itemize}
    Es gilt
    \begin{itemize}
        \item
            $+$ ist Assoziativiv:
            $\underset{a,b,c\in R}\forall (a+b)+c=a+(b+c)$
        \item
            $+$ ist Kommutativ:
            $\underset{a,b\in R}\forall a+b=b+a$
        \item
            $\cdot$ ist Assoziativiv:
            $\underset{a,b,c\in R}\forall
            (a\cdot b)\cdot c=a\cdot (b\cdot c)$
    \end{itemize}
    Weiter ist $0_R\neq 1_R$
    \begin{itemize}
        \item
            Neutrales Element der Addition:
            $\underset{a\in R}\forall a+0_R=a$
        \item
            Neutrales Element der Multiplikation:
            $\underset{a\in R}\forall a\cdot 1_R=a=1_R\cdot a$
    \end{itemize}
    Es ist außerdem
    \begin{itemize}
        \item
            Inverses der Addition: $\underset{a\in R}\forall
            \underset{-a\in R}-a+a=0$
    \end{itemize}
    Es gilt
    \begin{itemize}
        \item
            Distributivgesetz:
            $\underset{a,b,c\in R}\forall
            (a+b)\cdot c=a\cdot c+b\cdot c$
    \end{itemize}
    \color{blue}
    $R$ ist ein \textbf{Kommutativer Ring}, wenn zusätzlich
    $\underset{a,b\in R}\forall a\cdot b=b\cdot a$
    \color{black}

    Eine Teilmenge $S\subseteq R$ ist ein
    \textbf{Teilring (Unterring)} von $R$, wenn $0_R,1_R\in S$
    und unter den Operationen $+_R$ und $\cdot_R$
    abgeschlossen ist. Dann gilt
    $(S,+_R,\cdot_R,0_R,1_R)$ selber ein Ring.
\end{defi}

    \item

\begin{defi}
    Sei $R$ ein Integritätsbereich
    \begin{enumerate}[(a)]
        \item
            Ein invertierbares Element $a\in R$ heißt
            \textbf{Einheit}. Notation: $R^*\subseteq R$
            ist die Menge der Einheiten
        \item
            Zwei Elemente $a,b\in R$ sind
            \textbf{assoziiert}
            $a\sim b\Leftrightarrow a|b\land b|a$
        \item
            Ein Element $p\in R$ heißt \textbf{Primelement}
            (oder \textbf{prim}), falls $p\neq 0$ und
            $p\notin R^*$ und
            \begin{equation*}
                \underset{a,b\in R}\forall
                p|ab\Rightarrow p|a\lor p|b
            \end{equation*}
        \item
            Ein Element $u\in R$ heißt \textbf{Unzerlegbar}
            (oder \textbf{irreduzierbel}), falls $u\neq 0$
            und $u\notin R^*$ und
            \begin{equation*}
                \underset{a,b\in R}\forall u=a\cdot b
                \Rightarrow a\in R^*\lor b\in R^*
            \end{equation*}
    \end{enumerate}
\end{defi}

        \item
            Jedes Element $a\in R,a\neq 0$ ist ein
            endliches Produkt $a=u_1\cdot\hdots\cdot u_n$ von
            unzerlegbaren Elementen $u_i$,
            eindeutig bis auf Reihenfolge und Einheiten:
            Falls
            $a=u_1\cdot\hdots\cdot u_n=
            v_1\cdot\hdots\cdot v_l\in R$ unzerlegbar, dann
            ist $n=l$ und
            \begin{equation*}
                \underset{\pi\in\sum_n}\exists u_i\sim
                v_{\pi(i)}
            \end{equation*}
            Wobei $\sum_n$ die symmetrische Gruppe auf $n$
            Buchstaben.
        \item
            Jedes Element $a\in R,a\neq 0,a\notin R^*$
            ist ein endliches Produkt
            $a=u_1\cdot\hdots\cdot u_n$ von unzerlegbaren
            Elementen $u_1,\hdots,u_n$ und jedes unzerlegbare
            Element $u\in R$ ist Primelement.
    \end{enumerate}
\end{defi}
Beweis der Äquivalenz von (1) und (2):
\begin{itemize}
    \item
        $(1)\Rightarrow (2)$:
        Sei $u|ab\Rightarrow\underset{d}\exists ud=ab$
        z.z. $u|a$ oder $u|b$
        \begin{itemize}
            \item
                ist $a=0$ oder $b=0$, trivial
            \item
                ist $a$ eine Einheit:
                $u|a^{-1}(ab)=b$
            \item
                Sei nun $a,b\neq 0,a\notin R^*$
                Falls $d\in R^* ud=ab\Rightarrow
                u=(d^{-1}a)b$ Widerspruch,
                also $d\notin R^*$
        \end{itemize}
        Vorraussetzung $(1)\Rightarrow\exists$ Zerlegungen
        $a=p_1\cdot\hdots\cdot p_n,b=q_1\cdot\hdots\cdot
        q_l, d=v_1\cdot\hdots\cdot v_k$ jeweils Produkte
        aus unzerlegbaren Elementen. $ud=ab\Rightarrow
        uv_1\cdot\hdots\cdot v_k=p_1\cdot\hdots\cdot p_n
        q_1\cdot\hdots\cdot q_l\overset{(1)}\Rightarrow$
        eindeutig $\Rightarrow\underset{i}\exists u_\sim p_i
        \lor u\sim q_i\Rightarrow u|a\lor u|b$
    \item
        $(2)\Rightarrow (1)$ Sei $a=u_1\cdot\hdots\cdot u_n=
        v_1\cdot\hdots\cdot v_l$ (alle $u_i,v_j$ unzerlegbar)
        z.z. $n=l,\underset{\pi\in\sum_n}\exists u_i\sim
        v_{\pi(n)}$\\
        Induktion nach $n+l$, Induktionsanfang $n+l=1,n=l=1
        \Rightarrow a=u_1=v_1$\\
        Induktionsschritt $u_1|a=v_1\cdot\hdots\cdot v_l$.
        $u_1$ ist prim $\Rightarrow u_1$ teilt ein $v_j$
        $u_1|v_j=u_1\cdot s$ unzerlegbar also ist $s$ eine
        Einheit, $u_1\sim v_j$, $u_1=v_j\cdot s^{-1}$
        Nach kürzen folgt:
        \begin{equation*}
            u_2\cdot\hdots\cdot u_n=s^{-1}v_i\cdot\hdots
            \cdot v_{j-1}v_{j+1}\cdot\hdots\cdot v_l
        \end{equation*}
        Anzahl Faktoren $n-1,l-1\rightarrow n+l-2$
\end{itemize}
Eindeutige Primfaktorzerlegung bestimmt $\ggT$ und $\kgV$
und Teilbarkeit:\\
Gegeben $st,\in\R^*,a,b:a=sp_1^{\alpha_1}\hdots
p_n^{\alpha_n}, b=tp_1^{\beta_1}\hdots p_n^{\beta_n}$ mit
$\alpha_i, \beta_j\neq 0,p_1,\hdots,p_n$ paarweise nicht
assoziierte Primelemente
$a|b\Leftrightarrow\underset{i}\forall\alpha_i\leq\beta_i$,
\begin{itemize}
    \item
        $\ggT(a,b)\sim p_1^{m_1}p_2^{m_2}\hdots
        p_n^{m_n}$ mit $m_i=\min(\alpha_i,\beta_i)$
    \item
        $\kgV(a,b)\sim p_1^{M_1}p_2^{M_2}\hdots
        p_n^{M_n}$ mit $M_i=\max(\alpha_i,\beta_i)$
    \item
        $a|bc$ mit $b$ und $c$ teilerfremd, d.h.
        $\ggT(b,a)=1\Rightarrow a|c$
\end{itemize}
