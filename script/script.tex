\begin{thm}[Descantes]
    Sei $f(x)=a_nx^n+a_{n-1}x^{n-1}+\hdots+a_1x+a_0\in\Z[x]$\\
    und sei $x_0=\frac{r}{s}$ eine Lösung von $f(x)=0$.
    Dann ist $r\in\Z$ ein Teiler von $a_0$ und $s\in\Z$
    ist ein Teiler von $a_n$\\
    Beweis: Sei $x_0=\frac{r}{s}$, $f(x_0)=0$
    \begin{align*}
        f(x_0)&=0\\
        \Rightarrow
        a_n\left(\frac{r}{s}\right)^n+\hdots+a_0&=0&|\cdot s^n\\
        \Rightarrow
        a_nr^n+a_{n-1}r^{n-1}s+\hdots+a_1rs^{n-1}+a_0s^n&=0\\
        \Rightarrow
        r(a_nr^{n-1}+\hdots+a_1s^{n-1})&=-a_0s^n
    \end{align*}
    $r,s$ sind Teilerfremd, aber
    \begin{equation*}
        (r|-a_0s^n)\Rightarrow r|a_0
    \end{equation*}
    $s$ ausklammern:
    \begin{equation*}
        (s|-a_nr^n)\Rightarrow s|a_n
    \end{equation*}
\end{thm}

\begin{defi}
    Ein \textbf{Ring} $R$ ist ein Quintupel $(R,+,\cdot,0,1)$
    bestehend aus eine Menge $R$ mit zwei Abbildungen
    \begin{itemize}
        \item
            Addition:
               $+:R\times R\rightarrow R,(a,b)\mapsto a+b$
        \item
            Multiplikation:
               $\cdot:R\times R\rightarrow R,(a,b)
            \mapsto a\cdot b$
    \end{itemize}
    Es gilt
    \begin{itemize}
        \item
            $+$ ist Assoziativiv:
            $\underset{a,b,c\in R}\forall (a+b)+c=a+(b+c)$
        \item
            $+$ ist Kommutativ:
            $\underset{a,b\in R}\forall a+b=b+a$
        \item
            $\cdot$ ist Assoziativiv:
            $\underset{a,b,c\in R}\forall
            (a\cdot b)\cdot c=a\cdot (b\cdot c)$
    \end{itemize}
    Weiter ist $0_R\neq 1_R$
    \begin{itemize}
        \item
            Neutrales Element der Addition:
            $\underset{a\in R}\forall a+0_R=a$
        \item
            Neutrales Element der Multiplikation:
            $\underset{a\in R}\forall a\cdot 1_R=a=1_R\cdot a$
    \end{itemize}
    Es ist außerdem
    \begin{itemize}
        \item
            Inverses der Addition: $\underset{a\in R}\forall
            \underset{-a\in R}-a+a=0$
    \end{itemize}
    Es gilt
    \begin{itemize}
        \item
            Distributivgesetz:
            $\underset{a,b,c\in R}\forall
            (a+b)\cdot c=a\cdot c+b\cdot c$
    \end{itemize}
    \color{blue}
    $R$ ist ein \textbf{Kommutativer Ring}, wenn zusätzlich
    $\underset{a,b\in R}\forall a\cdot b=b\cdot a$
    \color{black}

    Eine Teilmenge $S\subseteq R$ ist ein
    \textbf{Teilring (Unterring)} von $R$, wenn $0_R,1_R\in S$
    und unter den Operationen $+_R$ und $\cdot_R$
    abgeschlossen ist. Dann gilt
    $(S,+_R,\cdot_R,0_R,1_R)$ selber ein Ring.
\end{defi}

    \item

    \varphi(a\cdot_R b)=\varphi(a)\cdot_S\varphi(b)$

\end{align*}

            \begin{equation*}
                \underset{a,b\in R}\forall u=a\cdot b
                \Rightarrow a\in R^*\lor b\in R^*
            \end{equation*}
    \end{enumerate}

\begin{defi}
    Sei $R$ ein Integritätsbereich.
    $R$ ist ein \textbf{Faktorieller Ring}, wenn
    gilt:
    \begin{enumerate}[(1)]
        \item
            Jedes Element $a\in R,a\neq 0$ ist ein
            endliches Produkt $a=u_1\cdot\hdots\cdot u_n$ von
            unzerlegbaren Elementen $u_i$,
            eindeutig bis auf Reihenfolge und Einheiten:
            Falls
            $a=u_1\cdot\hdots\cdot u_n=
            v_1\cdot\hdots\cdot v_l\in R$ unzerlegbar, dann
            ist $n=l$ und
            \begin{equation*}
                \underset{\pi\in\sum_n}\exists u_i\sim
                v_{\pi(i)}
            \end{equation*}
            Wobei $\sum_n$ die symmetrische Gruppe auf $n$
            Buchstaben.
        \item
            Jedes Element $a\in R,a\neq 0,a\notin R^*$
            ist ein endliches Produkt
            $a=u_1\cdot\hdots\cdot u_n$ von unzerlegbaren
            Elementen $u_1,\hdots,u_n$ und jedes unzerlegbare
            Element $u\in R$ ist Primelement.
    \end{enumerate}
\end{defi}
Beweis der Äquivalenz von (1) und (2):
\begin{itemize}
    \item
        $(1)\Rightarrow (2)$:
        Sei $u|ab\Rightarrow\underset{d}\exists ud=ab$
        z.z. $u|a$ oder $u|b$
        \begin{itemize}
            \item
                ist $a=0$ oder $b=0$, trivial
            \item
                ist $a$ eine Einheit:
                $u|a^{-1}(ab)=b$
            \item
                Sei nun $a,b\neq 0,a\notin R^*$
                Falls $d\in R^* ud=ab\Rightarrow
                u=(d^{-1}a)b$ Widerspruch,
                also $d\notin R^*$
        \end{itemize}
        Vorraussetzung $(1)\Rightarrow\exists$ Zerlegungen
        $a=p_1\cdot\hdots\cdot p_n,b=q_1\cdot\hdots\cdot
        q_l, d=v_1\cdot\hdots\cdot v_k$ jeweils Produkte
        aus unzerlegbaren Elementen. $ud=ab\Rightarrow
        uv_1\cdot\hdots\cdot v_k=p_1\cdot\hdots\cdot p_n
        q_1\cdot\hdots\cdot q_l\overset{(1)}\Rightarrow$
        eindeutig $\Rightarrow\underset{i}\exists u_\sim p_i
        \lor u\sim q_i\Rightarrow u|a\lor u|b$
    \item
        $(2)\Rightarrow (1)$ Sei $a=u_1\cdot\hdots\cdot u_n=
        v_1\cdot\hdots\cdot v_l$ (alle $u_i,v_j$ unzerlegbar)
        z.z. $n=l,\underset{\pi\in\sum_n}\exists u_i\sim
        v_{\pi(n)}$\\
        Induktion nach $n+l$, Induktionsanfang $n+l=1,n=l=1
        \Rightarrow a=u_1=v_1$\\
        Induktionsschritt $u_1|a=v_1\cdot\hdots\cdot v_l$.
        $u_1$ ist prim $\Rightarrow u_1$ teilt ein $v_j$
        $u_1|v_j=u_1\cdot s$ unzerlegbar also ist $s$ eine
        Einheit, $u_1\sim v_j$, $u_1=v_j\cdot s^{-1}$
        Nach kürzen folgt:
        \begin{equation*}
            u_2\cdot\hdots\cdot u_n=s^{-1}v_i\cdot\hdots
            \cdot v_{j-1}v_{j+1}\cdot\hdots\cdot v_l
        \end{equation*}
        Anzahl Faktoren $n-1,l-1\rightarrow n+l-2$
\end{itemize}
Eindeutige Primfaktorzerlegung bestimmt $\ggT$ und $\kgV$
und Teilbarkeit:\\
Gegeben $st,\in\R^*,a,b:a=sp_1^{\alpha_1}\hdots

    \item

    und Rechtsideal ist ($I\unlhd R$).\\
\end{defi}

\begin{proposition}
    Sei $R$ ein kommutativer Ring. Dann sind die folgenden
    Bedingungen Äquivalent
    \begin{enumerate}[(a)]
        \item
            Jede aufsteigende Kette $I_0\subset I_1\hdots$
            von aufsteigenden Idealen in $R$ wird stationär.
        \item
            In jeder nicht leeren Menge von Idealen in $R$
            gibt es ein bezüglich Inklusion maximales
            Element.
        \item
            Jedes Ideal in $R$ ist endlich erzeugt, d.h.
            von der Form $\langle a_1,\hdots,a_n\rangle
            =Ra_1+\hdots+Ra_n$ mit $a_i\in R$.
    \end{enumerate}
    Wenn die Aussagen erfüllt sind, dann ist $R$ ein
    \textbf{noetherscher Ring}.
\end{proposition}
Beweis:
\begin{itemize}
    \item $(a)\Rightarrow (b)$:\\
        Sei $M$ eine nicht leere Menge von Idealen. Falls es
        kein bzgl. inklusion maximales Element gibt,
        existiert eine nicht stationär werdende aufsteigende
        Kette $I_1\subsetneq I_2\subsetneq\hdots\subsetneq
        I_n\subsetneq I_{n+1}\subsetneq\hdots$.
    \item $(b)\Rightarrow (c)$:\\
        Sei $I$ Ideal,
        \begin{equation*}
            M=\{J\unlhd R|J\subseteq I, J
            \text{ endlich erzeugt}\}
        \end{equation*}
        (Hauptideale sind von einem Element erzeugt)
        $\Rightarrow M\neq\emptyset\Rightarrow$ es existiert
        maximales Element
        \begin{itemize}
            \item
                Fall $J=I$
            \item
                Fall $J\neq I\Rightarrow J\subsetneq I
                \Rightarrow\underset{x\in I}\exists
                x\notin J, J\in M$ endlich erzeugt
                \begin{equation*}
                    J\subsetneq (J+Rx)\subset I
                \end{equation*}
                Widerspruch zu $J$ maximal.
        \end{itemize}
        Nur der Fall $J=I$ kann eintreten.
    \item $(c)\Rightarrow (a)$:\\
        Sei $I_0\subseteq I_1\subseteq\hdots$ eine
        aufsteigende Kette von Idealen.
        $J:=\underset{n\in\N_0}\bigcup I_n$ ist ein
        Ideal: $a,b\in J\Rightarrow\underset{n_1}\exists
        a\in I_{n_1},\underset{n_2}\exists b\in I_{n_2},
        n_3:=\max\{n_1,n_2\}\Rightarrow a,b\in I_{n_3}$
        Ideal $\Rightarrow a+b\in I_{n_3}
        \Rightarrow$ abgeschlossen unter Addition.\\
        (c)$\Rightarrow J$ ist endlich erzeugt $\Rightarrow
        \underset{a_1,\hdots,a_n}\exists J=\langle
        a_1,\hdots,a_n\rangle=(a_1,\hdots,a_m),
        a_1\in I_{n_1},\hdots,a_m\in I_{n_m},N:=\max

        \hdots\subset I_{n_N}\subset I_n\Rightarrow
        a_1\hdots,a_m\in I_N\Rightarrow J=I_N\Rightarrow$
        Die kette ist stationär.
\end{itemize}

\begin{defi}
    Eine \textbf{Gruppe} ist ein Tripel $(G,*,e)$, bestehend
    aus einer Menge $G$, einer Abbildung
    $*:G\times G\rightarrow G$ und einem asugezeichneten
    Element $e=1_G$, sodass gilt
    \begin{itemize}
        \item $\underset{g_1,g_2,g_3\in G}\forall
            (g_1*g_2)*g_3=g_1*(g_2*g_3)$
        \item $\underset{g\in G}\forall g*e=1=e*g$
        \item $\underset{g\in G}\forall\underset{g^{-1}\in G}
            \exists g*g^{-1}=e=g^{-1}*g$
    \end{itemize}
    $G$ ist eine \textbf{endliche Gruppe}, wenn die Menge
    $G$ endlich ist.\\
    $G$ ist eine \textbf{abelsche Gruppe}, wenn $*$
    kommutativ ist.
\end{defi}

\begin{thm}
    Sei $G$ eine Gruppe.
    \begin{enumerate}[(a)]
        \item Eine Untergruppe $H\subset G$ ist normal
            $\Leftrightarrow \underset{g\in G}\forall
            gH^{-1}g=H\Leftrightarrow\underset{g\in G,h\in H}
            \forall gh^{-1}g\in H$
        \item
            Sei $N\triangleleft G$ und $G|N:=\{gN|g\in G\}$
            Die Menge der Linksnebenklassen. Dann ist $G|N$
            eine Gruppe mit Multiplikation $g_1N*g_2N=
            (g_1*g_2)N$. $G/N$ heißt \textbf{Faktorgruppe}
            oder \textbf{Quotientengruppe}.
        \item Sei $\varphi:G\to G'$ ein
            Gruppenhomomorphismus und $H:=\Kern(\varphi):=
            \{g\in G|\varphi(g)=e_G\}$. Dann ist $H$ ein
            Normalteiler. Außerdem ist $\varphi(G)$ eine
            Untergruppe von $G'$ und $\varphi(G)\cong G|H$.
            Wenn $\varphi$ surjektiv ist, gilt $G'\cong G|H$.
    \end{enumerate}
\end{thm}
